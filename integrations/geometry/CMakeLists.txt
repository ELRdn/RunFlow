cmake_minimum_required(VERSION 3.24)
project(runflow_geometry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(Boost_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake/boost" CACHE PATH "Pinned header-only Boost adapter" FORCE)
find_package(CGAL 6.2.1 EXACT REQUIRED)
add_executable(runflow_inspect inspect.cpp)
add_executable(runflow_autorefine autorefine.cpp)
target_link_libraries(runflow_inspect PRIVATE CGAL::CGAL)
target_link_libraries(runflow_autorefine PRIVATE CGAL::CGAL)
if(MSVC)
  target_compile_options(runflow_inspect PRIVATE /bigobj /fp:precise)
  target_compile_options(runflow_autorefine PRIVATE /bigobj /fp:precise)
endif()
